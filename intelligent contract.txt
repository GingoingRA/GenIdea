from genlayer import *
import json
import typing

class GenIdea(gl.Contract):
    usernames: TreeMap[Address, str]
    user_idea_count: TreeMap[Address, u256]
    idea_users: TreeMap[u256, Address]
    idea_texts: TreeMap[u256, str]
    idea_scores: TreeMap[u256, u256]
    idea_evaluated: TreeMap[u256, bool]
    next_idea_id: u256
    
    def __init__(self):
        self.next_idea_id = u256(1)
    
    @gl.public.write
    def set_username(self, username: str) -> None:
        if len(username) < 2 or len(username) > 30:
            raise Exception("Username must be 2-30 characters")
        self.usernames[gl.message.sender_address] = username
    
    @gl.public.write
    def submit_idea(self, idea_text: str) -> None:
        user = gl.message.sender_address
        
        if user not in self.usernames:
            raise Exception("Please set username first")
        
        current_count = self.user_idea_count.get(user, u256(0))
        if current_count >= u256(3):
            raise Exception("Maximum 3 ideas per user")
        
        if len(idea_text) < 20 or len(idea_text) > 500:
            raise Exception("Idea must be 20-500 characters")
        
        idea_id = self.next_idea_id
        self.idea_users[idea_id] = user
        self.idea_texts[idea_id] = idea_text
        self.idea_scores[idea_id] = u256(0)
        self.idea_evaluated[idea_id] = False
        self.user_idea_count[user] = current_count + u256(1)
        self.next_idea_id = self.next_idea_id + u256(1)
    
    @gl.public.write
    def score_idea(self, idea_id: int) -> None:
        idea_id_u256 = u256(idea_id)
        
        if idea_id_u256 not in self.idea_texts:
            raise Exception("Idea not found")
        
        if self.idea_evaluated.get(idea_id_u256, False):
            raise Exception("Idea already scored")
        
        idea_text = self.idea_texts[idea_id_u256]
        
        # Escape quotes in idea text to avoid breaking JSON
        safe_text = idea_text.replace('"', "'").replace('\n', ' ')
        
        prompt = f"""Rate 1-10: {safe_text}

1-3=weak 4-6=average 7-8=good 9-10=excellent

Return JSON: {{"score": 7}}"""
        
        def get_score():
            result = gl.nondet.exec_prompt(prompt)
            cleaned = result.strip()
            
            # Remove markdown
            if "```" in cleaned:
                parts = cleaned.split("```")
                for part in parts:
                    if "{" in part:
                        cleaned = part
                        break
            
            # Clean up
            cleaned = cleaned.replace("json", "").replace("JSON", "").strip()
            
            # Parse JSON - with fallback
            score_val = 5
            if "{" in cleaned and "}" in cleaned:
                parsed = json.loads(cleaned)
                score_val = int(parsed.get("score", 5))
            
            # Validate range
            if score_val < 1:
                score_val = 1
            if score_val > 10:
                score_val = 10
            
            return score_val
        
        score_result = gl.eq_principle.strict_eq(get_score)
        self.idea_scores[idea_id_u256] = u256(score_result * 10)
        self.idea_evaluated[idea_id_u256] = True
    
    @gl.public.view
    def get_leaderboard(self) -> dict[str, typing.Any]:
        ideas_list = []
        
        for idea_id in range(1, int(self.next_idea_id)):
            idea_id_u256 = u256(idea_id)
            
            if idea_id_u256 in self.idea_texts and idea_id_u256 in self.idea_users:
                user_addr = self.idea_users[idea_id_u256]
                username = self.usernames.get(user_addr, "Anonymous")
                
                ideas_list.append({
                    "id": idea_id,
                    "username": username,
                    "idea": self.idea_texts[idea_id_u256],
                    "score": int(self.idea_scores.get(idea_id_u256, u256(0))),
                    "evaluated": self.idea_evaluated.get(idea_id_u256, False)
                })
        
        # Simple bubble sort
        n = len(ideas_list)
        for i in range(n):
            for j in range(0, n - i - 1):
                if ideas_list[j]["score"] < ideas_list[j + 1]["score"]:
                    temp = ideas_list[j]
                    ideas_list[j] = ideas_list[j + 1]
                    ideas_list[j + 1] = temp
        
        return {"total_ideas": n, "ideas": ideas_list}
    
    @gl.public.view
    def get_my_ideas(self) -> dict[str, typing.Any]:
        user = gl.message.sender_address
        my_ideas = []
        
        for idea_id in range(1, int(self.next_idea_id)):
            idea_id_u256 = u256(idea_id)
            if idea_id_u256 in self.idea_users:
                if self.idea_users[idea_id_u256] == user:
                    my_ideas.append({
                        "id": idea_id,
                        "idea": self.idea_texts[idea_id_u256],
                        "score": int(self.idea_scores.get(idea_id_u256, u256(0))),
                        "evaluated": self.idea_evaluated.get(idea_id_u256, False)
                    })
        
        return {
            "username": self.usernames.get(user, ""),
            "ideas_submitted": len(my_ideas),
            "ideas_remaining": max(0, 3 - len(my_ideas)),
            "my_ideas": my_ideas
        }
    
    @gl.public.view
    def get_my_idea_ids(self) -> dict[str, typing.Any]:
        user = gl.message.sender_address
        my_ids = []
        
        for idea_id in range(1, int(self.next_idea_id)):
            idea_id_u256 = u256(idea_id)
            if idea_id_u256 in self.idea_users:
                if self.idea_users[idea_id_u256] == user:
                    my_ids.append({
                        "id": idea_id,
                        "idea": self.idea_texts[idea_id_u256],
                        "evaluated": self.idea_evaluated.get(idea_id_u256, False)
                    })
        
        return {
            "username": self.usernames.get(user, ""),
            "my_idea_ids": my_ids
        }
    
    @gl.public.write
    def score_my_next_idea(self) -> None:
        user = gl.message.sender_address
        
        # Find first unscored idea by this user
        for idea_id in range(1, int(self.next_idea_id)):
            idea_id_u256 = u256(idea_id)
            if idea_id_u256 in self.idea_users:
                if self.idea_users[idea_id_u256] == user:
                    if not self.idea_evaluated.get(idea_id_u256, False):
                        # Score this idea
                        self.score_idea(idea_id)
                        return
        
        raise Exception("No unscored ideas found for this user")
